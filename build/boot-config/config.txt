# CPZ-Alpine v0.1 — config.txt (console profile)
#
# Deltas from the M5 stock image (2026-06-08, sha256 1b1002f7...):
#   - removed: kernel=u-boot.bin  → direct kernel8.img boot (the direction
#     of M5's current pi-gen branch; one less moving part)
#   - changed: dtparam=audio=on → off (CPZ audio is the ES8389 codec,
#     not the firmware PWM audio path)
#   - added:   gpu_mem=32 (proven on this board by M. Birth; console
#     profile from PLAN §6)
# Everything else is faithful to stock, one-change-at-a-time discipline.

dtparam=i2c_arm=on
dtparam=i2s=on
dtparam=spi=on
dtparam=audio=off

camera_auto_detect=0
display_auto_detect=1
auto_initramfs=1

# Enable DRM VC4 V3D driver (HDMI)
dtoverlay=vc4-kms-v3d
max_framebuffers=2

disable_fw_kms_setup=1
arm_64bit=1
disable_overscan=1
arm_boost=1

# Camera (standard model; probes silently on Lite)
dtoverlay=imx219

[cm4]
otg_mode=1

[cm5]
dtoverlay=dwc2,dr_mode=host

[all]
# USB dwc2 mode (v0.12.1): HOST — GL852G hub up, USB-A port live,
# SR9900 ethernet enumerates (r8152 driver, ID added per mainline lore
# patch 2026-06-13).
#
# DEAD END (v0.12.0, do not retry): dr_mode=otg. On this board dwc2
# gets no role-detection input — with otg it registers as the g_serial
# gadget ("bound driver g_serial", UDC "not attached") and NEVER
# switches to host: no hub, no ethernet, no USB-A. The kernel exposes
# no runtime role switch either (/sys/class/usb_role/ is empty; dwc2
# debugfs has no writable mode file). One controller, one role.
#
# If the USB-C serial gadget is ever needed again: set dr_mode=peripheral
# here and re-enable the ttyGS0 getty in etc/inittab + g_serial in
# /etc/modules. (Costs ethernet + USB-A while set.)
dtoverlay=dwc2,dr_mode=host

# Console GPU memory profile (VideoCore)
gpu_mem=32

# UART exposed on the expansion bus; serial0 console in cmdline.txt
enable_uart=1

# CardputerZero board support (legacy combined overlay = stock-shipped DTBO)
dtoverlay=cardputerzero-overlay
dtoverlay=camera-gpio16-high-overlay
dtoverlay=bq27220
dtoverlay=gpio-ir,gpio_pin=13,gpio_pull=up
dtoverlay=pwm-ir-tx,gpio_pin=12,func=4
dtoverlay=spk-gpio24-high-overlay

# v0.22.1: BMM150 magnetometer (compass) behind the BMI270 auxiliary I2C bus.
# The overlay creates the magnetometer platform device; the bmc150_magn_bmi270
# module (in extra/) probes it and exposes /sys/bus/iio/devices/iio:device2.
dtoverlay=bmi270_bmm150_overlay

# v0.18.0: HDMI must stay dark until the on-demand desktop starts
# (cpz-desktop / the HDMI app). No firmware rainbow splash, no warnings.
disable_splash=1
avoid_warnings=1

# v0.22.3: disable on-board Bluetooth to free PL011 UART for GPIO 14/15.
# The CardputerZero has only two UARTs: PL011 (ttyAMA0) and mini-UART
# (ttyS0).  The PL011 can talk to EITHER the Murata BT chip (on internal
# traces) OR GPIO 14/15 (expansion port, where the Cap LoRa-1262 hat
# connects), but not both at once.  disable-bt picks the LoRa path.
#
# If you need Bluetooth back: remove the disable-bt and uart0 lines,
# reboot, and the LoRa hat goes dark.  A launcher toggle (similar to
# Tailscale/HDMI) that writes /boot/config.txt and reboots is the
# planned UX for switching between them.
dtoverlay=disable-bt
dtoverlay=uart0
